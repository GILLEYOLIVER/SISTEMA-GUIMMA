# 1) Limpa
taskkill /F /IM INSTAL_BANCO.exe 2>$null
Remove-Item -Recurse -Force .\RELEASE -ErrorAction SilentlyContinue
Remove-Item -Recurse -Force .\build -ErrorAction SilentlyContinue
Remove-Item -Recurse -Force .\dist -ErrorAction SilentlyContinue
Remove-Item -Force .\INSTAL_BANCO.spec -ErrorAction SilentlyContinue

# 2) Compila já montando a pasta final com tudo dentro

  --name "INSTAL_BANCO" `
  --distpath ".\RELEASE" `
  --add-data "config.json;." `
  --add-data "FUNDO_INSTALADOR.jpg;." `

  main.py
